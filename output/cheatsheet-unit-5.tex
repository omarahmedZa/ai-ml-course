\sect{MDP}
$(S,A,P,R,\gamma)$; \ $\pi(a\mid s)$; \ $G_t=\sum_{k\ge0}\gamma^k r_{t+k+1}$\\
Horizon $1/(1-\gamma)$: $\gamma{=}0.9\Rightarrow10$, $0.99\Rightarrow100$

\sect{Values}
$V^\pi(s)=\sum_a\pi(a\mid s)\sum_{s'}P(s'\mid s,a)[R+\gamma V^\pi(s')]$\\
$Q^\pi(s,a)=\sum_{s'}P[R+\gamma V^\pi(s')]$ \quad $V^*=\max_a Q^*$\\
TD: $V\leftarrow V+\alpha[r+\gamma V(s')-V(s)]$; bracket $=$ TD error\\
Terminal state: $V=0$, never bootstrap off it.

\sect{Exploration}
$\varepsilon$-greedy: $P(\text{best})=1-\varepsilon+\varepsilon/k$\\
UCB: $\arg\max_a\left[Q(a)+c\sqrt{\ln t/N(a)}\right]$\\
Regret $=T\cdot r^*-\sum_t r_t$; good algorithms $O(\log T)$

$Q\leftarrow Q+\alpha[r+\gamma\max_{a'}Q(s',a')-Q]$

\sect{Gotchas}
\begin{itemize}
\item Reward is the spec: agents optimise exactly what you wrote.
\item Greedy on noisy estimates locks onto the wrong arm forever.
\item $\gamma{=}0$ is myopia, not efficiency.
\end{itemize}
