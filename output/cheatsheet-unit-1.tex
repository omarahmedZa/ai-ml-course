$X\in\mathbb{R}^{n\times d}$,\ $y\in\mathbb{R}^{n}$,\ $f_\theta$,\ $L$\\

\sect{Linear \& logistic}
$\hat y=w\cdot x+b$ \quad closed form $(X^\top X)^{-1}X^\top y$\\
$\sigma(z)=\dfrac{1}{1+e^{-z}}$ \quad $z=\log\dfrac{p}{1-p}$\\
Boundary: $w\cdot x+b=0$ \ (always a hyperplane)\\
$\mathrm{BCE}=-[y\log p+(1-y)\log(1-p)]$\\
Gradient: $\partial L/\partial w=(p-y)x$

\sect{Trees \& ensembles}
$\mathrm{Gini}=1-\sum_k p_k^2$ \quad $H=-\sum_k p_k\log p_k$\\
Bagging $=$ variance; \ boosting $=$ bias.\\
Trees cannot extrapolate: a leaf predicts a training average.

\sect{Gotchas}
\begin{itemize}
\item Accuracy on 1\% positives: 99\% by predicting ``no''.
\item Scaling before the split leaks test statistics.
\item Impurity importance favours high-cardinality features.
\end{itemize}
