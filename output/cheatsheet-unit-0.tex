\sect{Multiplication: the only operation}
$x\cdot w=\sum_i x_i w_i=\|x\|\|w\|\cos\theta$
$\cos\theta=\dfrac{x\cdot w}{\|x\|\|w\|}\in[-1,1]$
$(m\times n)(n\times k)\to(m\times k)$ \quad inner dims must match\\
$Wx$ = $m$ dot products at once; \ $xW\neq Wx$

\sect{Derivatives}
$\dfrac{d}{dw}w^n=nw^{n-1}$ \quad $\dfrac{d}{dw}(aw+b)=a$\\
Chain: $\dfrac{d}{dw}f(g(w))=f'(g(w))g'(w)$\\
Update: $w\leftarrow w-\eta\nabla f$ \ (minus: gradient is uphill)

\sect{Probability}
$E[X]=\sum_x x\,p(x)$ \quad $\mathrm{Var}=E[(X-EX)^2]$\\
$L(\theta)=p(\text{data}\mid\theta)$ \quad $\mathrm{NLL}=-\sum_i\log p_\theta(y_i\mid x_i)$\\
$\mathrm{softmax}(z)_i=\dfrac{e^{z_i}}{\sum_j e^{z_j}}$ \quad $\text{perplexity}=e^{\mathrm{loss}}$
