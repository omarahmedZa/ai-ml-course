\sect{Forward pass}
$z^\ell=W^\ell a^{\ell-1}+b^\ell$ \quad $a^\ell=\varphi(z^\ell)$\\
$W^\ell\in\mathbb{R}^{n_\ell\times n_{\ell-1}}$ \quad params $=n_\ell n_{\ell-1}+n_\ell$\\
Batch: $Z^\ell=A^{\ell-1}(W^\ell)^\top+b^\ell$, shape $B\times n_\ell$

\sect{Why the bend is mandatory}
$W_2(W_1x+b_1)+b_2=(W_2W_1)x+(W_2b_1+b_2)$\\
$\Rightarrow$ any depth collapses to one matrix without $\varphi$.

\sect{Backprop}
$\dfrac{\partial L}{\partial w}=\dfrac{\partial L}{\partial \text{out}}\cdot\dfrac{\partial \text{out}}{\partial z}\cdot\dfrac{\partial z}{\partial w}$\\
Fan-out: gradients \textbf{add}. Reverse topological order, one sweep.

\sect{Training loop}
\texttt{pred = model(x); loss = crit(pred,y)}\\
\texttt{loss.backward(); opt.step(); opt.zero\_grad()}\\
steps $=$ epochs $\times N/B$ \quad horizon: lr low $\to$ plateau

\sect{Gotchas}
\begin{itemize}
\item Skip \texttt{zero\_grad} $\Rightarrow$ gradients accumulate.
\item Eval with dropout on $\Rightarrow$ noisy, wrong metric.
\item Dead ReLU: gradient 0 forever. LeakyReLU/GELU or lower lr.
\end{itemize}
