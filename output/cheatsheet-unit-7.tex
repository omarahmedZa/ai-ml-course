\sect{Attention}
$\mathrm{Attn}(Q,K,V)=\mathrm{softmax}\!\left(\dfrac{QK^\top}{\sqrt{d_k}}\right)V$\\
$Q$ asks, $K$ advertises, $V$ delivers. Cost $O(n^2d)$.\\
Scale by $\sqrt{d_k}$: score std grows like $\sqrt{d_k}$; unscaled softmax saturates.\\
Causal mask: set $j>i$ to $-\infty$ before softmax.

\sect{Block}
$x\leftarrow x+\mathrm{MHA}(\mathrm{LN}(x))$; \ $x\leftarrow x+\mathrm{FFN}(\mathrm{LN}(x))$\\
Params/block at $d$: $4d^2$ (attention) $+8d^2$ (FFN) $=12d^2$\\
Position must be injected; attention alone is permutation-invariant.

\sect{Pretraining}
$\mathrm{NLL}=-\sum_t\log p_\theta(t_t\mid t_{<t})$; \ perplexity $=e^{\mathrm{NLL}}$\\
GPT causal (generation) \ vs \ BERT masked 15\% (encoding)\\
loss 3.0 nats $\Rightarrow$ perplexity 20.

\sect{Alignment}
SFT $\to$ reward model (Bradley--Terry) $\to$ PPO $+\,\beta\,\mathrm{KL}$\\
DPO: same preference optimum, no reward model, no RL.\\
Sampling: $p=\mathrm{softmax}(z/T)$; $T\downarrow$ sharpens, adds no knowledge.

\sect{Gotchas}
\begin{itemize}
\item Attention weights $\neq$ explanation.
\item Higher temperature $\neq$ creativity; it raises rare tokens.
\item Perplexity $\neq$ instruction following.
\end{itemize}
