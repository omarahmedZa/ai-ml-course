def gini(labels):
    """Gini impurity of a node: 1 - sum(p_k ** 2)."""
    n = len(labels)
    if n == 0:
        return 0.0
    counts = {}
    for v in labels:
        counts[v] = counts.get(v, 0) + 1
    return 1.0 - sum((c / n) ** 2 for c in counts.values())
