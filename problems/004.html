def empirical_risk(preds, targets):
    """Average squared error over the dataset (the empirical risk)."""
    if len(preds) != len(targets):
        raise ValueError("lengths differ")
    return float(sum((p - y) ** 2 for p, y in zip(preds, targets)))  # BUG: total, not average
